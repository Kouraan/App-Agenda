// INICIAR AS TABELAS 
-- Utilizador (autenticação fica no Supabase Auth, 
-- esta tabela é para os dados da app)
CREATE TABLE IF NOT EXISTS utilizador (
    id       SERIAL PRIMARY KEY,
    nome     TEXT NOT NULL,
    password TEXT NOT NULL
);

-- Clientes
CREATE TABLE IF NOT EXISTS clientes (
    id               SERIAL PRIMARY KEY,
    nome             TEXT NOT NULL UNIQUE,
    numero_telefone  TEXT NOT NULL,
    tipo_cliente     TEXT NOT NULL DEFAULT 'NORMAL',
    faltas           INTEGER NOT NULL DEFAULT 0,
    dia_semana       TEXT,
    hora_corte       TEXT,
    rapido           INTEGER NOT NULL DEFAULT 0
);

-- Marcações
CREATE TABLE IF NOT EXISTS marcacoes (
    id           SERIAL PRIMARY KEY,
    data_hora    TEXT NOT NULL UNIQUE,
    cliente_nome TEXT NOT NULL,
    duracao      INTEGER NOT NULL DEFAULT 30,
    observacoes  TEXT NOT NULL DEFAULT '',
    falta        INTEGER NOT NULL DEFAULT 0
);

-- Anotações
CREATE TABLE IF NOT EXISTS anotacoes (
    id    INTEGER PRIMARY KEY DEFAULT 1,
    texto TEXT NOT NULL DEFAULT '',
    CONSTRAINT apenas_uma_linha CHECK (id = 1)
);

-- Pendentes
CREATE TABLE IF NOT EXISTS pendentes (
    id               SERIAL PRIMARY KEY,
    nome             TEXT NOT NULL,
    numero_telefone  TEXT NOT NULL DEFAULT ''
);

-- Logs
CREATE TABLE IF NOT EXISTS logs (
    id        SERIAL PRIMARY KEY,
    data_hora TEXT NOT NULL,
    tipo      TEXT NOT NULL,
    mensagem  TEXT NOT NULL
);

-- Fila de sync offline
CREATE TABLE IF NOT EXISTS sync_pendente (
    id        SERIAL PRIMARY KEY,
    operacao  TEXT NOT NULL,
    tabela    TEXT NOT NULL,
    dados     TEXT NOT NULL,
    criado_em TEXT NOT NULL
);


// DAR PERMISSAO DE LEITURA E ESCRITA

//  Colunas updated_at + triggers automáticos
alter table public.clientes  add column if not exists updated_at timestamptz not null default now();
alter table public.marcacoes add column if not exists updated_at timestamptz not null default now();
alter table public.pendentes add column if not exists updated_at timestamptz not null default now();
alter table public.anotacoes add column if not exists updated_at timestamptz not null default now();

create or replace function public.set_updated_at()
returns trigger as $$
begin
  new.updated_at = now();
  return new;
end;
$$ language plpgsql;

create trigger trg_clientes_updated_at  before update on public.clientes  for each row execute function public.set_updated_at();
create trigger trg_marcacoes_updated_at before update on public.marcacoes for each row execute function public.set_updated_at();

// Tabela de Tombstones
create table if not exists public.sync_tombstones (
    id         bigserial primary key,
    tabela     text not null,
    chave      text not null,
    apagado_em timestamptz not null default now()
);

alter table public.sync_tombstones enable row level security;
create policy "acesso_autenticado" on public.sync_tombstones
    for all to authenticated using (true) with check (true);
grant select, insert, update, delete on public.sync_tombstones to authenticated;
grant usage, select on all sequences in schema public to authenticated;

// CORRIGIR CONFLITOS

drop trigger if exists trg_clientes_key_change on public.clientes;
drop trigger if exists trg_marcacoes_key_change on public.marcacoes;
drop function if exists public.tombstone_on_key_change();

------------------

create or replace function public.tombstone_cliente_renomeado()
returns trigger as $$
begin
  if OLD.nome is distinct from NEW.nome then
    insert into public.sync_tombstones (tabela, chave) values ('clientes', OLD.nome);
  end if;
  return new;
end;
$$ language plpgsql;

create or replace function public.tombstone_marcacao_remarcada()
returns trigger as $$
begin
  if OLD.data_hora is distinct from NEW.data_hora then
    insert into public.sync_tombstones (tabela, chave) values ('marcacoes', OLD.data_hora);
  end if;
  return new;
end;
$$ language plpgsql;

create trigger trg_clientes_key_change
  after update on public.clientes
  for each row execute function public.tombstone_cliente_renomeado();

create trigger trg_marcacoes_key_change
  after update on public.marcacoes
  for each row execute function public.tombstone_marcacao_remarcada();
